OOP · Herança e Abstração

Classe Abstrata: Imovel

from abc import ABC, abstractmethod from dataclasses import dataclass @dataclass class Imovel(ABC): valor_base: float qtd_quartos: int = 1 tem_garagem: bool = False def adicionar_garagem(self) -> float: """Acréscimo fixo de R$300 para casa e apartamento.""" return 300.0 if self.tem_garagem else 0.0 @abstractmethod def calcular_aluguel(self, cliente_sem_criancas: bool = False) -> float: """Cada subclasse implementa suas próprias regras.""" ...
@dataclass elimina boilerplate do __init__. @abstractmethod garante que toda subclasse implemente calcular_aluguel.

OOP · Polimorfismo

Herança: Apartamento e Casa

@dataclass class Apartamento(Imovel): valor_base: float = 700.0 def calcular_aluguel( self, sem_criancas: bool = False ) -> float: total = self.valor_base if self.qtd_quartos == 2: total += 200.0 total += self.adicionar_garagem() if sem_criancas: total *= 0.95 # -5% return round(total, 2)
@dataclass class Casa(Imovel): valor_base: float = 900.0 def calcular_aluguel( self, sem_criancas: bool = False ) -> float: total = self.valor_base if self.qtd_quartos == 2: total += 250.0 total += self.adicionar_garagem() return round(total, 2) # Casa 2 quartos + garagem: # 900 + 250 + 300 = R$ 1.450,00

OOP · Regra especial

Herança: Estudio (garagem diferenciada)

@dataclass class Estudio(Imovel): valor_base: float = 1200.0 qtd_vagas_extra: int = 0 # vagas além das 2 primeiras def calc_vagas_estudio(self) -> float: """2 vagas = R$250; cada vaga extra = +R$60.""" if not self.tem_garagem: return 0.0 return 250.0 + (self.qtd_vagas_extra * 60.0) def calcular_aluguel(self, sem_criancas: bool = False) -> float: return round(self.valor_base + self.calc_vagas_estudio(), 2) # Estudio c/ garagem 2 vagas + 1 extra: 1200 + 250 + 60 = R$ 1.510,00
Polimorfismo em ação: mesma chamada calcular_aluguel(), comportamentos distintos por tipo de imóvel.

OOP · Composição

Orcamento e Exportação CSV

import csv, io from dataclasses import dataclass @dataclass class Orcamento: cliente_nome: str cliente_sem_criancas: bool imovel: Imovel taxa_contrato: float = 2000.0 parcelas_contrato: int = 1 def gerar_mensal(self) -> float: aluguel = self.imovel.calcular_aluguel(self.cliente_sem_criancas) parcela = self.taxa_contrato / self.parcelas_contrato return round(aluguel + parcela, 2) def exportar_csv(self) -> str: output = io.StringIO() writer = csv.writer(output) writer.writerow(["Mês", "Valor (R$)"]) mensal = self.gerar_mensal() for mes in range(1, 13): writer.writerow([mes, f"{mensal:.2f}"]) return output.getvalue()

Camada UI · Streamlit

Conectando tudo com Streamlit

import streamlit as st st.title("🏠 Imobiliária R.M — Orçamento de Aluguel") tipo = st.selectbox("Tipo de imóvel", ["Apartamento", "Casa", "Estudio"]) quartos = st.number_input("Quartos", min_value=1, max_value=4) garagem = st.checkbox("Incluir garagem?") sem_criancas = st.checkbox("Cliente sem crianças? (desconto apto)") parcelas = st.slider("Parcelas do contrato", 1, 5, 1) nome = st.text_input("Nome do cliente") imovel = {"Apartamento": Apartamento, "Casa": Casa, "Estudio": Estudio}[tipo]( qtd_quartos=quartos, tem_garagem=garagem ) orc = Orcamento(nome, sem_criancas, imovel, parcelas_contrato=parcelas) if st.button("Gerar Orçamento"): st.metric("Valor mensal", f"R$ {orc.gerar_mensal():.2f}") st.download_button("⬇ Baixar CSV", orc.exportar_csv(), "orcamento.csv")

Recapitulando

Pontos-chave

Herança

Apartamento, Casa e Estudio herdam de Imovel — compartilham estrutura, divergem em comportamento.

Polimorfismo

Uma chamada calcular_aluguel() produz resultados diferentes dependendo do tipo real do objeto.

Composição

Orcamento agrega Imovel, Cliente e Contrato — sem herança. Tem-um, não é-um.

UI separada do domínio

Streamlit só lê inputs e exibe outputs — não conhece nenhuma regra de negócio.
